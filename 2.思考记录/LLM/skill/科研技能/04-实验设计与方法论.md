## 样本量计算

### 为什么需要样本量计算
```
样本量过小:
├── 统计功效不足 (β错误)
├── 无法检测真实效应
└── 资源浪费

样本量过大:
├── 伦理问题
├── 资源浪费
└── 微小效应也可能显著
```

### 常用公式

**两独立样本t检验**
```python
from scipy.stats import norm

def sample_size_ttest(d, alpha=0.05, power=0.80):
    """计算两独立样本t检验所需样本量
    d: Cohen's d效应量
    alpha: 显著性水平
    power: 统计功效 (1-β)
    """
    z_alpha = norm.ppf(1 - alpha/2)
    z_beta = norm.ppf(power)
    n = 2 * ((z_alpha + z_beta) / d) ** 2
    return int(np.ceil(n))

# 示例：检测中等效应 (d=0.5)
n = sample_size_ttest(d=0.5)  # 每组约64人
```

**相关系数检验**
```python
def sample_size_correlation(r, alpha=0.05, power=0.80):
    """计算相关系数检验所需样本量
    r: 预期相关系数
    """
    z_alpha = norm.ppf(1 - alpha/2)
    z_beta = norm.ppf(power)
    # Fisher's z变换
    z_r = 0.5 * np.log((1+r)/(1-r))
    n = ((z_alpha + z_beta) / z_r) ** 2 + 3
    return int(np.ceil(n))
```

## 随机化方法

### 简单随机化
```python
import random

def simple_randomization(n, groups=2):
    """简单随机化"""
    assignments = [random.choice(range(groups)) for _ in range(n)]
    return assignments

# 示例：100人分两组
assignments = simple_randomization(100)
```

### 区组随机化
```python
def block_randomization(n, block_size=4, groups=2):
    """区组随机化：保证各组人数均衡"""
    assignments = []
    while len(assignments) < n:
        block = list(range(groups)) * (block_size // groups)
        random.shuffle(block)
        assignments.extend(block)
    return assignments[:n]
```

### 分层随机化
```python
def stratified_randomization(df, stratify_cols, group_col='group'):
    """分层随机化：平衡协变量"""
    df[group_col] = ''
    for _, group_df in df.groupby(stratify_cols):
        indices = group_df.index.tolist()
        random.shuffle(indices)
        half = len(indices) // 2
        df.loc[indices[:half], group_col] = 'treatment'
        df.loc[indices[half:], group_col] = 'control'
    return df
```

## 常见实验设计

### 2×2因子设计
```python
import statsmodels.api as sm
from statsmodels.formula.api import ols

# 模拟数据
np.random.seed(42)
n = 100
factor_a = np.repeat(['A1', 'A2'], n//2)
factor_b = np.tile(['B1', 'B2'], n//2)
y = 50 + 3*(factor_a=='A2') + 2*(factor_b=='B2') + \
    1.5*(factor_a=='A2')*(factor_b=='B2') + np.random.normal(0, 5, n)

df = pd.DataFrame({'A': factor_a, 'B': factor_b, 'y': y})

# 双因素方差分析
model = ols('y ~ C(A) * C(B)', data=df).fit()
anova_table = sm.stats.anova_lm(model, typ=2)
print(anova_table)
```

### 交叉设计
```python
def crossover_design(n_subjects, n_periods=2, n_treatments=2):
    """交叉设计：每个受试者接受所有处理"""
    # Williams设计平衡顺序效应
    sequences = []
    treatments = list(range(n_treatments))
    for i in range(n_subjects):
        seq = treatments.copy()
        if i % 2 == 1:
            seq = seq[::-1]  # 反序平衡
        sequences.append(seq)
    return sequences
```
