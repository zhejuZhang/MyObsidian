# 动态地图与实时建图

> 动态地图与实时建图是自动驾驶系统适应动态环境的关键技术，实现地图的实时更新和维护。

---

## 一、基本概念

### 1.1 动态地图定义
动态地图是能够实时反映环境变化的地图，包括：
- **静态信息**：道路几何、车道线等
- **准静态信息**：交通标志、信号灯等
- **动态信息**：交通流、障碍物等

### 1.2 实时建图定义
实时建图是利用传感器数据在线构建和更新环境地图的过程。

### 1.3 在自动驾驶中的作用

#### 环境感知
- **实时环境建模**：实时建模当前环境
- **变化检测**：检测环境变化
- **动态物体过滤**：过滤动态物体

#### 决策支持
- **路径规划**：为路径规划提供最新信息
- **风险评估**：评估潜在风险
- **行为预测**：预测交通参与者行为

## 二、在线建图技术

### 2.1 增量式建图

#### 增量式点云建图
```python
class IncrementalPointCloudMapping:
    """
    增量式点云建图
    """
    def __init__(self):
        self.global_map = PointCloud()
        self.local_map = PointCloud()
        self.registration = ICPRegistration()
    
    def add_frame(self, point_cloud, pose):
        """
        添加一帧点云
        """
        # 变换到全局坐标系
        transformed = self.transform(point_cloud, pose)
        
        # 配准到全局地图
        if self.global_map.size() > 0:
            transformation = self.registration.register(
                transformed, self.global_map
            )
            transformed = self.apply_transformation(transformed, transformation)
        
        # 更新局部地图
        self.update_local_map(transformed, pose)
        
        # 融合到全局地图
        self.global_map.merge(transformed)
        
        # 地图优化
        self.optimize_map()
    
    def update_local_map(self, new_points, pose):
        """
        更新局部地图
        """
        # 移除远处的点
        self.local_map = self.local_map.crop_sphere(pose, radius=100)
        
        # 添加新点
        self.local_map.merge(new_points)
```

### 2.2 滑动窗口建图

#### 滑动窗口地图
```python
class SlidingWindowMapping:
    """
    滑动窗口建图
    """
    def __init__(self, window_size=10):
        self.window_size = window_size
        self.frames = []
        self.map = PointCloud()
    
    def add_frame(self, point_cloud, pose):
        """
        添加一帧
        """
        # 添加到窗口
        self.frames.append({
            'point_cloud': point_cloud,
            'pose': pose
        })
        
        # 保持窗口大小
        if len(self.frames) > self.window_size:
            self.frames.pop(0)
        
        # 重建局部地图
        self.rebuild_local_map()
    
    def rebuild_local_map(self):
        """
        重建局部地图
        """
        self.map = PointCloud()
        
        for frame in self.frames:
            # 变换到全局坐标系
            transformed = self.transform(
                frame['point_cloud'], frame['pose']
            )
            self.map.merge(transformed)
```

### 2.3 基于关键帧的建图

#### 关键帧地图
```python
class KeyframeBasedMapping:
    """
    基于关键帧的建图
    """
    def __init__(self):
        self.keyframes = []
        self.map = PointCloud()
        self.keyframe_selector = KeyframeSelector()
    
    def add_frame(self, point_cloud, pose):
        """
        添加一帧
        """
        # 判断是否为关键帧
        if self.keyframe_selector.is_keyframe(pose, self.keyframes):
            # 添加关键帧
            self.keyframes.append({
                'point_cloud': point_cloud,
                'pose': pose
            })
            
            # 更新地图
            self.update_map()
    
    def update_map(self):
        """
        更新地图
        """
        self.map = PointCloud()
        
        for keyframe in self.keyframes:
            # 变换到全局坐标系
            transformed = self.transform(
                keyframe['point_cloud'], keyframe['pose']
            )
            self.map.merge(transformed)
```

## 三、动态物体过滤

### 3.1 基于多帧的过滤

#### 多帧一致性检查
```python
class MultiFrameConsistencyFilter:
    """
    多帧一致性过滤
    """
    def __init__(self):
        self.history_frames = []
    
    def filter_dynamic_objects(self, current_frame, current_pose):
        """
        过滤动态物体
        """
        # 添加到历史帧
        self.history_frames.append({
            'frame': current_frame,
            'pose': current_pose
        })
        
        # 保持历史帧数量
        if len(self.history_frames) > 10:
            self.history_frames.pop(0)
        
        # 检查一致性
        consistent_points = []
        dynamic_points = []
        
        for point in current_frame:
            if self.is_consistent(point, current_pose):
                consistent_points.append(point)
            else:
                dynamic_points.append(point)
        
        return consistent_points, dynamic_points
    
    def is_consistent(self, point, current_pose):
        """
        检查点的一致性
        """
        # 在历史帧中查找对应点
        for hist_frame in self.history_frames:
            # 变换到同一坐标系
            transformed_point = self.transform_point(
                point, current_pose, hist_frame['pose']
            )
            
            # 查找最近点
            nearest_point = self.find_nearest(
                transformed_point, hist_frame['frame']
            )
            
            # 检查距离
            if nearest_point is not None:
                distance = self.compute_distance(
                    transformed_point, nearest_point
                )
                if distance > 0.5:  # 阈值
                    return False
        
        return True
```

```python
class DeepDynamicObjectFilter:
    """
    深度学习动态物体过滤
    """
    def __init__(self):
        self.dynamic_detector = DynamicObjectDetector()
    
    def filter(self, point_cloud, image):
        """
        过滤动态物体
        """
        # 检测动态物体
        dynamic_objects = self.dynamic_detector.detect(point_cloud, image)
        
        # 创建动态物体掩码
        dynamic_mask = self.create_mask(dynamic_objects, point_cloud)
        
        # 过滤动态点
        static_points = point_cloud[~dynamic_mask]
        dynamic_points = point_cloud[dynamic_mask]
        
        return static_points, dynamic_points
```

```python
class SemanticBasedFilter:
    """
    基于语义的过滤
    """
    def __init__(self):
        self.semantic_segmentor = SemanticSegmentor()
        
        # 可能是动态物体的类别
        self.dynamic_classes = ['car', 'person', 'bicycle', 'motorcycle']
    
    def filter(self, point_cloud, image):
        """
        语义过滤
        """
        # 语义分割
        semantic_map = self.semantic_segmentor.segment(image)
        
        # 投影语义信息到点云
        point_semantics = self.project_semantics(point_cloud, semantic_map)
        
        # 过滤动态物体
        static_mask = ~np.isin(point_semantics, self.dynamic_classes)
        
        static_points = point_cloud[static_mask]
        dynamic_points = point_cloud[~static_mask]
        
        return static_points, dynamic_points
```

## 四、地图更新策略

### 4.1 增量更新

#### 增量地图更新
```python
class IncrementalMapUpdate:
    """
    增量地图更新
    """
    def __init__(self):
        self.map = None
        self.change_detector = ChangeDetector()
    
    def update(self, new_data, vehicle_pose):
        """
        增量更新地图
        """
        # 检测变化
        changes = self.change_detector.detect(
            new_data, self.map, vehicle_pose
        )
        
        # 应用变化
        for change in changes:
            if change['type'] == 'add':
                self.map.add(change['data'])
            elif change['type'] == 'modify':
                self.map.modify(change['id'], change['data'])
            elif change['type'] == 'delete':
                self.map.delete(change['id'])
        
        return changes
```

### 4.2 基于置信度的更新

#### 置信度地图更新
```python
class ConfidenceBasedUpdate:
    """
    基于置信度的地图更新
    """
    def __init__(self):
        self.map = ConfidenceMap()
    
    def update(self, new_data, confidence):
        """
        基于置信度更新
        """
        for i, point in enumerate(new_data):
            # 获取当前置信度
            current_confidence = self.map.get_confidence(point)
            
            # 更新置信度
            new_confidence = self.update_confidence(
                current_confidence, confidence[i]
            )
            
            # 更新地图
            if new_confidence > current_confidence:
                self.map.update(point, new_confidence)
    
    def update_confidence(self, old_confidence, new_confidence):
        """
        更新置信度
        """
        # 加权平均
        alpha = 0.7
        return alpha * old_confidence + (1 - alpha) * new_confidence
```

### 4.3 时序衰减更新

#### 时序衰减地图
```python
class TemporalDecayMap:
    """
    时序衰减地图
    """
    def __init__(self, decay_rate=0.95):
        self.map = {}
        self.decay_rate = decay_rate
        self.last_update = {}
    
    def update(self, new_data, timestamp):
        """
        时序衰减更新
        """
        # 更新新数据
        for point in new_data:
            key = self.get_key(point)
            self.map[key] = point
            self.last_update[key] = timestamp
        
        # 衰减旧数据
        self.decay_old_data(timestamp)
    
    def decay_old_data(self, current_time):
        """
        衰减旧数据
        """
        expired_keys = []
        
        for key, last_time in self.last_update.items():
            # 计算时间差
            time_diff = current_time - last_time
            
            # 应用衰减
            decay_factor = self.decay_rate ** time_diff
            
            # 如果衰减过多，标记删除
            if decay_factor < 0.1:
                expired_keys.append(key)
        
        # 删除过期数据
        for key in expired_keys:
            del self.map[key]
            del self.last_update[key]
```

### 5.1 多车协同建图

#### 协同地图构建
```python
class CollaborativeMapping:
    """
    多车协同建图
    """
    def __init__(self):
        self.vehicle_maps = {}
        self.global_map = PointCloud()
        self.map_fusion = MapFusion()
    
    def add_vehicle_data(self, vehicle_id, point_cloud, pose):
        """
        添加车辆数据
        """
        # 更新车辆局部地图
        if vehicle_id not in self.vehicle_maps:
            self.vehicle_maps[vehicle_id] = PointCloud()
        
        # 变换到全局坐标系
        transformed = self.transform(point_cloud, pose)
        self.vehicle_maps[vehicle_id].merge(transformed)
        
        # 融合到全局地图
        self.fuse_maps()
    
    def fuse_maps(self):
        """
        融合多车地图
        """
        self.global_map = self.map_fusion.fuse(
            list(self.vehicle_maps.values())
        )
```

```python
class CloudEdgeCollaborativeMapping:
    """
    云边协同建图
    """
    def __init__(self):
        self.edge_map = PointCloud()
        self.cloud_map = PointCloud()
        self.sync_manager = SyncManager()
    
    def edge_update(self, point_cloud, pose):
        """
        边缘端更新
        """
        # 本地更新
        transformed = self.transform(point_cloud, pose)
        self.edge_map.merge(transformed)

        云端更新

        edge_data = self.sync_manager.receive_from_edge()
        
        # 融合到云端地图
        self.cloud_map.merge(edge_data)
        
        # 下发更新
        self.sync_manager.send_to_edge(self.cloud_map)
```

## 六、地图一致性维护

### 6.1 一致性检查

#### 地图一致性检查
```python
class MapConsistencyChecker:
    """
    地图一致性检查
    """
    def check_consistency(self, map_data):
        """
        检查地图一致性
        """
        errors = []
        
        # 检查几何一致性
        geometric_errors = self.check_geometric_consistency(map_data)
        errors.extend(geometric_errors)
        
        # 检查拓扑一致性
        topology_errors = self.check_topology_consistency(map_data)
        errors.extend(topology_errors)
        
        # 检查语义一致性
        semantic_errors = self.check_semantic_consistency(map_data)
        errors.extend(semantic_errors)
        
        return errors
    
    def check_geometric_consistency(self, map_data):
        """
        检查几何一致性
        """
        errors = []
        
        # 检查重叠
        overlaps = self.detect_overlaps(map_data)
        if overlaps:
            errors.append({
                'type': 'geometric_overlap',
                'data': overlaps
            })
        
        # 检查缝隙
        gaps = self.detect_gaps(map_data)
        if gaps:
            errors.append({
                'type': 'geometric_gap',
                'data': gaps
            })
        
        return errors
```

```python
class ConsistencyRepairer:
    """
    一致性修复器
    """
    def repair(self, map_data, errors):
        """
        修复一致性问题
        """
        repaired_data = map_data.copy()
        
        for error in errors:
            if error['type'] == 'geometric_overlap':
                repaired_data = self.repair_overlap(repaired_data, error)
            elif error['type'] == 'geometric_gap':
                repaired_data = self.repair_gap(repaired_data, error)
            elif error['type'] == 'topology_error':
                repaired_data = self.repair_topology(repaired_data, error)
        
        return repaired_data
```
