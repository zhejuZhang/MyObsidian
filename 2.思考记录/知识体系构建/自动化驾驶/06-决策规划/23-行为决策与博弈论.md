# 行为决策与博弈论

> 行为决策与博弈论是自动驾驶实现智能决策的关键技术，处理复杂的交通场景和多智能体交互。

---

## 一、基本概念

### 1.1 行为决策定义
行为决策是自动驾驶系统根据当前环境状态选择合适驾驶行为的过程，包括：
- **纵向决策**：加速、减速、匀速
- **横向决策**：换道、保持车道、转弯
- **交互决策**：与其他交通参与者的交互

### 1.2 博弈论定义
博弈论是研究多个决策主体之间策略互动的数学理论，用于建模自动驾驶中的多智能体交互。

### 1.3 在自动驾驶中的作用

#### 决策支持
- **行为选择**：选择合适的驾驶行为
- **策略制定**：制定驾驶策略
- **风险评估**：评估决策风险

#### 交互处理
- **意图识别**：识别其他交通参与者意图
- **行为预测**：预测其他参与者行为
- **协同决策**：与其他参与者协同

## 二、马尔可夫决策过程（MDP）

### 2.1 MDP模型

#### MDP定义
```python
class MDP:
    """
    马尔可夫决策过程
    """
    def __init__(self):
        self.states = []  # 状态空间
        self.actions = []  # 动作空间
        self.transition_model = {}  # 转移模型
        self.reward_function = {}  # 奖励函数
        self.discount_factor = 0.9  # 折扣因子
    
    def get_transition_probability(self, state, action, next_state):
        """
        获取转移概率
        """
        return self.transition_model.get((state, action, next_state), 0)
    
    def get_reward(self, state, action, next_state):
        """
        获取奖励
        """
        return self.reward_function.get((state, action, next_state), 0)
```

### 2.2 值迭代算法

#### 值迭代
```python
class ValueIteration:
    """
    值迭代算法
    """
    def __init__(self, mdp):
        self.mdp = mdp
        self.values = {state: 0 for state in mdp.states}
    
    def iterate(self, max_iterations=1000, tolerance=1e-6):
        """
        值迭代
        """
        for i in range(max_iterations):
            new_values = {}
            delta = 0
            
            for state in self.mdp.states:
                # 计算每个动作的值
                action_values = []
                for action in self.mdp.actions:
                    value = self.calc_action_value(state, action)
                    action_values.append(value)
                
                # 取最大值
                new_values[state] = max(action_values)
                
                # 计算变化量
                delta = max(delta, abs(new_values[state] - self.values[state]))
            
            # 更新值函数
            self.values = new_values
            
            # 检查收敛
            if delta < tolerance:
                break
        
        return self.values
    
    def calc_action_value(self, state, action):
        """
        计算动作值
        """
        value = 0
        
        for next_state in self.mdp.states:
            prob = self.mdp.get_transition_probability(state, action, next_state)
            reward = self.mdp.get_reward(state, action, next_state)
            value += prob * (reward + self.mdp.discount_factor * self.values[next_state])
        
        return value
```

### 2.3 策略迭代算法

#### 策略迭代
```python
class PolicyIteration:
    """
    策略迭代算法
    """
    def __init__(self, mdp):
        self.mdp = mdp
        self.policy = {state: random.choice(mdp.actions) for state in mdp.states}
        self.values = {state: 0 for state in mdp.states}
    
    def iterate(self, max_iterations=100):
        """
        策略迭代
        """
        for i in range(max_iterations):
            # 策略评估
            self.evaluate_policy()
            
            # 策略改进
            if not self.improve_policy():
                break
        
        return self.policy
    
    def evaluate_policy(self):
        """
        策略评估
        """
        for _ in range(100):  # 迭代次数
            new_values = {}
            
            for state in self.mdp.states:
                action = self.policy[state]
                value = self.calc_state_value(state, action)
                new_values[state] = value
            
            self.values = new_values
    
    def improve_policy(self):
        """
        策略改进
        """
        policy_stable = True
        
        for state in self.mdp.states:
            old_action = self.policy[state]
            
            # 找到最优动作
            best_action = None
            best_value = -float('inf')
            
            for action in self.mdp.actions:
                value = self.calc_state_value(state, action)
                if value > best_value:
                    best_value = value
                    best_action = action
            
            # 更新策略
            self.policy[state] = best_action
            
            # 检查策略是否稳定
            if old_action != best_action:
                policy_stable = False
        
        return policy_stable
```

### 3.2 信念状态更新

#### 信念状态
```python
class BeliefState:
    """
    信念状态
    """
    def __init__(self, pomdp):
        self.pomdp = pomdp
        self.belief = {state: 1.0/len(pomdp.states) for state in pomdp.states}
    
    def update(self, action, observation):
        """
        更新信念状态
        """
        new_belief = {}
        
        for next_state in self.pomdp.states:
            # 计算先验
            prior = 0
            for state in self.pomdp.states:
                prob = self.pomdp.get_transition_probability(state, action, next_state)
                prior += prob * self.belief[state]
            
            # 计算似然
            likelihood = self.pomdp.get_observation_probability(
                next_state, action, observation
            )
            
            # 计算后验
            new_belief[next_state] = likelihood * prior
        
        # 归一化
        total = sum(new_belief.values())
        for state in new_belief:
            new_belief[state] /= total
        
        self.belief = new_belief
        
        return self.belief
```

## 四、博弈论模型

### 4.1 非合作博弈

#### 纳什均衡
```python
class NashEquilibrium:
    """
    纳什均衡
    """
    def __init__(self):
        self.players = []
        self.strategies = {}
        self.payoff_matrices = {}
    
    def add_player(self, player_id, strategies):
        """
        添加玩家
        """
        self.players.append(player_id)
        self.strategies[player_id] = strategies
    
    def set_payoff(self, player_id, strategy_profile, payoff):
        """
        设置收益
        """
        self.payoff_matrices[(player_id, strategy_profile)] = payoff
    
    def find_equilibrium(self):
        """
        寻找纳什均衡
        """
        equilibria = []
        
        # 遍历所有策略组合
        strategy_profiles = self.generate_strategy_profiles()
        
        for profile in strategy_profiles:
            if self.is_nash_equilibrium(profile):
                equilibria.append(profile)
        
        return equilibria
    
    def is_nash_equilibrium(self, profile):
        """
        检查是否为纳什均衡
        """
        for player_id in self.players:
            current_payoff = self.get_payoff(player_id, profile)
            
            # 检查是否有更好的策略
            for strategy in self.strategies[player_id]:
                if strategy != profile[player_id]:
                    # 计算偏离后的收益
                    deviated_profile = profile.copy()
                    deviated_profile[player_id] = strategy
                    deviated_payoff = self.get_payoff(player_id, deviated_profile)
                    
                    # 如果有更好策略，则不是纳什均衡
                    if deviated_payoff > current_payoff:
                        return False
        
        return True
```

### 4.2 合作博弈

#### 联盟博弈
```python
class CoalitionGame:
    """
    联盟博弈
    """
    def __init__(self):
        self.players = []
        self.coalition_values = {}
    
    def add_player(self, player_id):
        """
        添加玩家
        """
        self.players.append(player_id)
    
    def set_coalition_value(self, coalition, value):
        """
        设置联盟价值
        """
        self.coalition_values[frozenset(coalition)] = value
    
    def shapley_value(self):
        """
        计算夏普利值
        """
        shapley_values = {}
        
        for player in self.players:
            value = 0
            
            # 遍历所有不包含该玩家的联盟
            other_players = [p for p in self.players if p != player]
            for coalition_size in range(len(other_players) + 1):
                for coalition in combinations(other_players, coalition_size):
                    # 计算边际贡献
                    coalition_with = frozenset(coalition) | frozenset([player])
                    coalition_without = frozenset(coalition)
                    
                    marginal_contribution = (
                        self.coalition_values.get(coalition_with, 0) -
                        self.coalition_values.get(coalition_without, 0)
                    )
                    
                    # 加权
                    weight = (
                        math.factorial(len(coalition)) *
                        math.factorial(len(self.players) - len(coalition) - 1) /
                        math.factorial(len(self.players))
                    )
                    
                    value += weight * marginal_contribution
            
            shapley_values[player] = value
        
        return shapley_values
```

### 4.3 动态博弈

#### 斯塔克尔伯格博弈
```python
class StackelbergGame:
    """
    斯塔克尔伯格博弈
    """
    def __init__(self):
        self.leader = None
        self.follower = None
        self.leader_strategies = []
        self.follower_strategies = []
        self.payoff_function = None
    
    def set_leader(self, leader_id, strategies):
        """
        设置领导者
        """
        self.leader = leader_id
        self.leader_strategies = strategies
    
    def set_follower(self, follower_id, strategies):
        """
        设置跟随者
        """
        self.follower = follower_id
        self.follower_strategies = strategies
    
    def solve(self):
        """
        求解斯塔克尔伯格博弈
        """
        best_leader_strategy = None
        best_leader_payoff = -float('inf')
        
        # 领导者遍历所有策略
        for leader_strategy in self.leader_strategies:
            # 跟随者最优响应
            follower_best_response = self.follower_best_response(leader_strategy)
            
            # 计算领导者收益
            leader_payoff = self.payoff_function(
                self.leader, leader_strategy, follower_best_response
            )
            
            # 更新最优策略
            if leader_payoff > best_leader_payoff:
                best_leader_payoff = leader_payoff
                best_leader_strategy = leader_strategy
        
        return best_leader_strategy
    
    def follower_best_response(self, leader_strategy):
        """
        跟随者最优响应
        """
        best_strategy = None
        best_payoff = -float('inf')
        
        for follower_strategy in self.follower_strategies:
            payoff = self.payoff_function(
                self.follower, leader_strategy, follower_strategy
            )
            
            if payoff > best_payoff:
                best_payoff = payoff
                best_strategy = follower_strategy
        
        return best_strategy
```

## 五、强化学习决策

### 5.1 深度Q网络（DQN）

#### DQN算法
```python
class DQN:
    """
    深度Q网络
    """
    def __init__(self, state_dim, action_dim):
        self.state_dim = state_dim
        self.action_dim = action_dim
        
        # Q网络
        self.q_network = self.build_network()
        
        # 目标网络
        self.target_network = self.build_network()
        
        # 经验回放
        self.replay_buffer = ReplayBuffer()
    
    def build_network(self):
        """
        构建网络
        """
        model = nn.Sequential(
            nn.Linear(self.state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, self.action_dim)
        )
        return model
    
    def select_action(self, state, epsilon=0.1):
        """
        选择动作
        """
        if random.random() < epsilon:
            # 探索
            return random.randint(0, self.action_dim - 1)
        else:
            # 利用
            with torch.no_grad():
                state_tensor = torch.FloatTensor(state).unsqueeze(0)
                q_values = self.q_network(state_tensor)
                return q_values.argmax().item()
    
    def train(self, batch_size=32):
        """
        训练
        """
        # 采样经验
        batch = self.replay_buffer.sample(batch_size)
        
        # 计算目标Q值
        states, actions, rewards, next_states, dones = batch
        
        with torch.no_grad():
            next_q_values = self.target_network(next_states)
            max_next_q_values = next_q_values.max(dim=1)[0]
            target_q_values = rewards + (1 - dones) * 0.99 * max_next_q_values
        
        # 计算当前Q值
        current_q_values = self.q_network(states).gather(1, actions.unsqueeze(1))
        
        # 计算损失
        loss = nn.MSELoss()(current_q_values.squeeze(), target_q_values)
        
        # 更新网络
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
```

### 5.2 策略梯度方法

#### REINFORCE算法
```python
class REINFORCE:
    """
    REINFORCE算法
    """
    def __init__(self, state_dim, action_dim):
        self.state_dim = state_dim
        self.action_dim = action_dim
        
        # 策略网络
        self.policy_network = self.build_network()
        
        # 优化器
        self.optimizer = torch.optim.Adam(self.policy_network.parameters())
    
    def build_network(self):
        """
        构建网络
        """
        model = nn.Sequential(
            nn.Linear(self.state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, self.action_dim),
            nn.Softmax(dim=-1)
        )
        return model
    
    def select_action(self, state):
        """
        选择动作
        """
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        action_probs = self.policy_network(state_tensor)
        
        # 采样动作
        action_dist = torch.distributions.Categorical(action_probs)
        action = action_dist.sample()
        
        return action.item(), action_dist.log_prob(action)
    
    def update(self, log_probs, rewards):
        """
        更新策略
        """
        # 计算折扣回报
        returns = []
        R = 0
        for r in reversed(rewards):
            R = r + 0.99 * R
            returns.insert(0, R)
        returns = torch.FloatTensor(returns)
        
        # 标准化回报
        returns = (returns - returns.mean()) / (returns.std() + 1e-8)
        
        # 计算损失
        loss = 0
        for log_prob, R in zip(log_probs, returns):
            loss -= log_prob * R
        
        # 更新网络
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
```

### 5.3 近端策略优化（PPO）

#### PPO算法
```python
class PPO:
    """
    近端策略优化
    """
    def __init__(self, state_dim, action_dim):
        self.state_dim = state_dim
        self.action_dim = action_dim
        
        # 策略网络
        self.policy_network = self.build_network()
        
        # 价值网络
        self.value_network = self.build_network()
        
        # 优化器
        self.policy_optimizer = torch.optim.Adam(self.policy_network.parameters())
        self.value_optimizer = torch.optim.Adam(self.value_network.parameters())
    
    def select_action(self, state):
        """
        选择动作
        """
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        action_probs = self.policy_network(state_tensor)
        
        # 采样动作
        action_dist = torch.distributions.Categorical(action_probs)
        action = action_dist.sample()
        
        return action.item(), action_dist.log_prob(action)
    
    def update(self, states, actions, old_log_probs, returns, advantages):
        """
        更新策略
        """
        # 计算新策略的log概率
        action_probs = self.policy_network(states)
        action_dist = torch.distributions.Categorical(action_probs)
        new_log_probs = action_dist.log_prob(actions)
        
        # 计算比率
        ratio = torch.exp(new_log_probs - old_log_probs)
        
        # 计算裁剪目标
        clip_epsilon = 0.2
        clipped_ratio = torch.clamp(ratio, 1 - clip_epsilon, 1 + clip_epsilon)
        
        # 计算策略损失
        policy_loss = -torch.min(ratio * advantages, clipped_ratio * advantages).mean()
        
        # 计算价值损失
        values = self.value_network(states).squeeze()
        value_loss = nn.MSELoss()(values, returns)
        
        # 更新网络
        self.policy_optimizer.zero_grad()
        policy_loss.backward()
        self.policy_optimizer.step()
        
        self.value_optimizer.zero_grad()
        value_loss.backward()
        self.value_optimizer.step()
```
