# 局部路径规划与避障

> 局部路径规划与避障是自动驾驶实现安全行驶的关键技术，基于局部环境信息进行实时路径规划。

---

## 一、基本概念

### 1.1 局部规划定义
局部规划是在全局路径的基础上，基于局部环境信息进行实时路径规划，包括：
- **避障规划**：避开障碍物
- **轨迹规划**：生成可执行的轨迹
- **速度规划**：规划行驶速度

### 1.2 避障定义
避障是检测和避开行驶路径上的障碍物的过程。

### 1.3 在自动驾驶中的作用

#### 安全保障
- **碰撞避免**：避免与障碍物碰撞
- **安全距离**：保持安全距离
- **紧急避障**：紧急情况下的避障

#### 舒适性
- **平滑轨迹**：生成平滑的行驶轨迹
- **舒适加速度**：控制舒适的加速度
- **减少颠簸**：减少行驶颠簸

## 二、局部路径规划算法

### 2.1 动态窗口法（DWA）

#### DWA算法
```python
class DynamicWindowApproach:
    """
    动态窗口法
    """
    def __init__(self):
        self.max_speed = 1.0  # 最大速度
        self.min_speed = -0.5  # 最小速度
        self.max_yaw_rate = 40.0 * math.pi / 180.0  # 最大角速度
        self.max_accel = 0.2  # 最大加速度
        self.max_delta_yaw_rate = 40.0 * math.pi / 180.0  # 最大角加速度
        self.v_resolution = 0.01  # 速度分辨率
        self.yaw_rate_resolution = 0.1 * math.pi / 180.0  # 角速度分辨率
        self.dt = 0.1  # 时间步长
        self.predict_time = 3.0  # 预测时间
    
    def plan(self, current_state, goal, obstacles):
        """
        DWA规划
        """
        # 计算动态窗口
        dw = self.calc_dynamic_window(current_state)
        
        # 评估轨迹
        best_trajectory = None
        best_score = -float('inf')
        
        for v in np.arange(dw[0], dw[1], self.v_resolution):
            for yaw_rate in np.arange(dw[2], dw[3], self.yaw_rate_resolution):
                # 生成轨迹
                trajectory = self.generate_trajectory(
                    current_state, v, yaw_rate
                )
                
                # 评估轨迹
                score = self.evaluate_trajectory(trajectory, goal, obstacles)
                
                # 更新最佳轨迹
                if score > best_score:
                    best_score = score
                    best_trajectory = trajectory
        
        return best_trajectory
    
    def calc_dynamic_window(self, current_state):
        """
        计算动态窗口
        """
        # 速度窗口
        vs = [
            current_state.v - self.max_accel * self.dt,
            current_state.v + self.max_accel * self.dt
        ]
        
        # 角速度窗口
        yaw_rates = [
            current_state.yaw_rate - self.max_delta_yaw_rate * self.dt,
            current_state.yaw_rate + self.max_delta_yaw_rate * self.dt
        ]
        
        # 限制在最大最小范围内
        vs[0] = max(vs[0], self.min_speed)
        vs[1] = min(vs[1], self.max_speed)
        yaw_rates[0] = max(yaw_rates[0], -self.max_yaw_rate)
        yaw_rates[1] = min(yaw_rates[1], self.max_yaw_rate)
        
        return [vs[0], vs[1], yaw_rates[0], yaw_rates[1]]
```

### 2.2 人工势场法

#### 势场法算法
```python
class ArtificialPotentialField:
    """
    人工势场法
    """
    def __init__(self):
        self.k_att = 1.0  # 引力增益
        self.k_rep = 100.0  # 斥力增益
        self.d0 = 2.0  # 斥力影响范围
    
    def plan(self, current_pos, goal, obstacles):
        """
        势场法规划
        """
        # 计算引力
        f_att = self.calc_attractive_force(current_pos, goal)
        
        # 计算斥力
        f_rep = self.calc_repulsive_force(current_pos, obstacles)
        
        # 合力
        f_total = f_att + f_rep
        
        # 计算下一位置
        next_pos = current_pos + f_total * 0.1
        
        return next_pos
    
    def calc_attractive_force(self, current_pos, goal):
        """
        计算引力
        """
        # 引力与距离成正比
        force = self.k_att * (goal - current_pos)
        return force
    
    def calc_repulsive_force(self, current_pos, obstacles):
        """
        计算斥力
        """
        force = np.zeros(2)
        
        for obstacle in obstacles:
            # 计算距离
            dist = np.linalg.norm(current_pos - obstacle)
            
            # 斥力与距离平方成反比
            if dist < self.d0:
                rep_force = self.k_rep * (1.0/dist - 1.0/self.d0) * (1.0/dist**2)
                direction = (current_pos - obstacle) / dist
                force += rep_force * direction
        
        return force
```

### 2.3 时空联合规划

#### 时空规划算法
```python
class SpatioTemporalPlanner:
    """
    时空联合规划器
    """
    def __init__(self):
        self.time_horizon = 5.0  # 时间范围
        self.dt = 0.1  # 时间步长
        self.lateral_resolution = 0.5  # 横向分辨率
        self.longitudinal_resolution = 1.0  # 纵向分辨率
    
    def plan(self, current_state, obstacles, predictions):
        """
        时空规划
        """
        # 生成候选轨迹
        candidate_trajectories = self.generate_candidates(current_state)
        
        # 评估轨迹
        scored_trajectories = []
        for trajectory in candidate_trajectories:
            score = self.evaluate_trajectory(
                trajectory, obstacles, predictions
            )
            scored_trajectories.append((score, trajectory))
        
        # 选择最佳轨迹
        best_trajectory = max(scored_trajectories, key=lambda x: x[0])[1]
        
        return best_trajectory
    
    def generate_candidates(self, current_state):
        """
        生成候选轨迹
        """
        candidates = []
        
        # 横向采样
        for lateral_offset in np.arange(-3, 3, self.lateral_resolution):
            # 纵向采样
            for longitudinal_offset in np.arange(0, 20, self.longitudinal_resolution):
                # 生成轨迹
                trajectory = self.generate_trajectory(
                    current_state, lateral_offset, longitudinal_offset
                )
                candidates.append(trajectory)
        
        return candidates
```

### 3.2 障碍物预测

#### 运动预测
```python
class ObstaclePredictor:
    """
    障碍物预测器
    """
    def __init__(self):
        self.prediction_horizon = 3.0  # 预测时间范围
        self.dt = 0.1  # 时间步长
    
    def predict(self, obstacles, current_time):
        """
        预测障碍物运动
        """
        predictions = {}
        
        for obstacle in obstacles:
            # 预测轨迹
            predicted_trajectory = self.predict_trajectory(
                obstacle, current_time
            )
            predictions[obstacle.id] = predicted_trajectory
        
        return predictions
    
    def predict_trajectory(self, obstacle, current_time):
        """
        预测单个障碍物轨迹
        """
        trajectory = []
        
        # 匀速直线预测
        for t in np.arange(0, self.prediction_horizon, self.dt):
            future_time = current_time + t
            future_position = obstacle.position + obstacle.velocity * t
            
            trajectory.append({
                'time': future_time,
                'position': future_position
            })
        
        return trajectory
```

### 3.3 碰撞检测

#### 碰撞检测器
```python
class CollisionDetector:
    """
    碰撞检测器
    """
    def __init__(self):
        self.safety_margin = 0.5  # 安全裕度
    
    def check_collision(self, trajectory, obstacles):
        """
        检测碰撞
        """
        for point in trajectory:
            for obstacle in obstacles:
                # 计算距离
                distance = self.compute_distance(
                    point['position'], obstacle.position
                )
                
                # 检查是否碰撞
                if distance < self.safety_margin + obstacle.radius:
                    return True, point, obstacle
        
        return False, None, None
    
    def compute_distance(self, pos1, pos2):
        """
        计算距离
        """
        return np.linalg.norm(pos1 - pos2)
```

### 4.1 轨迹平滑

#### 轨迹平滑器
```python
class TrajectorySmoother:
    """
    轨迹平滑器
    """
    def __init__(self):
        self.smoothing_weight = 0.5
        self.obstacle_weight = 1.0
        self.curvature_weight = 0.1
    
    def smooth(self, trajectory, obstacles):
        """
        平滑轨迹
        """
        smoothed = trajectory.copy()
        
        for _ in range(100):  # 迭代次数
            # 平滑项
            smooth_gradient = self.calc_smooth_gradient(smoothed)
            
            # 障碍物项
            obstacle_gradient = self.calc_obstacle_gradient(smoothed, obstacles)
            
            # 曲率项
            curvature_gradient = self.calc_curvature_gradient(smoothed)
            
            # 更新轨迹
            gradient = (
                self.smoothing_weight * smooth_gradient +
                self.obstacle_weight * obstacle_gradient +
                self.curvature_weight * curvature_gradient
            )
            
            smoothed = smoothed - 0.01 * gradient
        
        return smoothed
```

### 4.2 速度规划

#### 速度规划器
```python
class VelocityPlanner:
    """
    速度规划器
    """
    def __init__(self):
        self.max_speed = 15.0  # 最大速度（m/s）
        self.max_acceleration = 2.0  # 最大加速度（m/s²）
        self.max_deceleration = 4.0  # 最大减速度（m/s²）
    
    def plan(self, path, obstacles, current_speed):
        """
        规划速度
        """
        velocities = []
        
        for i, point in enumerate(path):
            # 计算限制速度
            limit_speed = self.calc_limit_speed(point, obstacles)
            
            # 计算期望速度
            desired_speed = min(limit_speed, self.max_speed)
            
            # 考虑加速度限制
            if i > 0:
                max_speed_change = self.max_acceleration * 0.1
                desired_speed = min(
                    desired_speed,
                    velocities[-1] + max_speed_change
                )
            
            velocities.append(desired_speed)
        
        return velocities
    
    def calc_limit_speed(self, point, obstacles):
        """
        计算限制速度
        """
        min_speed = self.max_speed
        
        for obstacle in obstacles:
            # 计算距离
            distance = self.compute_distance(point, obstacle)
            
            # 根据距离限制速度
            if distance < 5.0:
                limit_speed = distance * 2.0  # 简单的速度限制
                min_speed = min(min_speed, limit_speed)
        
        return min_speed
```

## 五、紧急避障

### 5.1 紧急制动

#### 紧急制动控制器
```python
class EmergencyBrakeController:
    """
    紧急制动控制器
    """
    def __init__(self):
        self.emergency_deceleration = 8.0  # 紧急减速度
        self.warning_distance = 10.0  # 预警距离
        self.brake_distance = 5.0  # 制动距离
    
    def check_emergency(self, current_speed, distance_to_obstacle):
        """
        检查是否需要紧急制动
        """
        # 计算制动距离
        stopping_distance = current_speed ** 2 / (2 * self.emergency_deceleration)
        
        # 判断是否需要紧急制动
        if distance_to_obstacle < stopping_distance + 1.0:
            return True
        
        return False
    
    def control(self, current_speed, distance_to_obstacle):
        """
        紧急制动控制
        """
        # 计算所需减速度
        required_deceleration = current_speed ** 2 / (2 * distance_to_obstacle)
        
        # 限制在最大减速度内
        deceleration = min(required_deceleration, self.emergency_deceleration)
        
        return -deceleration
```

```python
class EmergencySteerController:
    """
    紧急转向控制器
    """
    def __init__(self):
        self.max_steer_angle = 45.0  # 最大转向角
        self.steer_rate_limit = 30.0  # 转向速率限制
    
    def check_emergency(self, current_state, obstacle):
        """
        检查是否需要紧急转向
        """
        # 判断是否可以通过转向避障
        if self.can_avoid_by_steer(current_state, obstacle):
            return True
        
        return False
    
    def control(self, current_state, target_direction):
        """
        紧急转向控制
        """
        # 计算目标转向角
        target_steer = self.calc_steer_angle(
            current_state.heading, target_direction
        )
        
        # 限制转向速率
        steer_change = target_steer - current_state.steer_angle
        steer_change = np.clip(
            steer_change,
            -self.steer_rate_limit * 0.1,
            self.steer_rate_limit * 0.1
        )
        
        # 计算转向命令
        steer_command = current_state.steer_angle + steer_change
        
        return np.clip(steer_command, -self.max_steer_angle, self.max_steer_angle)
```

## 六、多目标优化

### 6.1 代价函数设计

#### 多目标代价函数
```python
class MultiObjectiveCostFunction:
    """
    多目标代价函数
    """
    def __init__(self):
        self.weights = {
            'smoothness': 0.3,
            'safety': 0.4,
            'efficiency': 0.2,
            'comfort': 0.1
        }
    
    def evaluate(self, trajectory, goal, obstacles):
        """
        评估轨迹
        """
        # 计算各项代价
        smoothness_cost = self.calc_smoothness_cost(trajectory)
        safety_cost = self.calc_safety_cost(trajectory, obstacles)
        efficiency_cost = self.calc_efficiency_cost(trajectory, goal)
        comfort_cost = self.calc_comfort_cost(trajectory)
        
        # 加权求和
        total_cost = (
            self.weights['smoothness'] * smoothness_cost +
            self.weights['safety'] * safety_cost +
            self.weights['efficiency'] * efficiency_cost +
            self.weights['comfort'] * comfort_cost
        )
        
        return total_cost
```

### 6.2 帕累托优化

#### 帕累托最优
```python
class ParetoOptimizer:
    """
    帕累托优化器
    """
    def optimize(self, candidate_trajectories):
        """
        帕累托优化
        """
        pareto_set = []
        
        for trajectory in candidate_trajectories:
            # 检查是否被支配
            dominated = False
            for pareto_trajectory in pareto_set:
                if self.dominates(pareto_trajectory, trajectory):
                    dominated = True
                    break
            
            # 如果不被支配，加入帕累托集
            if not dominated:
                # 移除被当前轨迹支配的轨迹
                pareto_set = [
                    t for t in pareto_set
                    if not self.dominates(trajectory, t)
                ]
                pareto_set.append(trajectory)
        
        return pareto_set
```
