**自动文献综述**
```python
import requests
from datetime import datetime, timedelta

def auto_literature_review(topic, days=7):
    """自动检索近期相关文献"""
    # Semantic Scholar API
    url = "https://api.semanticscholar.org/graph/v1/paper/search"
    params = {
        "query": topic,
        "limit": 50,
        "fields": "title,authors,year,citationCount,abstract",
        "year": f"{datetime.now().year}-{datetime.now().year}"
    }
    
    response = requests.get(url, params=params)
    papers = response.json()['data']
    
    # 按引用量排序
    papers.sort(key=lambda x: x['citationCount'], reverse=True)
    
    # 生成综述摘要
    summary = f"## {topic} 近期文献综述\n\n"
    for paper in papers[:10]:
        summary += f"### {paper['title']}\n"
        summary += f"- 作者: {', '.join([a['name'] for a in paper['authors'][:3]])}\n"
        summary += f"- 引用: {paper['citationCount']}\n"
        summary += f"- 摘要: {paper['abstract'][:200]}...\n\n"
    
    return summary
```

### Windows任务计划
```powershell
# 创建定时备份任务
$action = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-File C:\scripts\backup.ps1"
$trigger = New-ScheduledTaskTrigger -Daily -At 8am
Register-ScheduledTask -TaskName "ObsidianBackup" -Action $action -Trigger $trigger
```

### Claude Code集成
```python
# 使用Claude分析Obsidian笔记
import anthropic

client = anthropic.Anthropic()

def analyze_notes(notes_content):
    """使用Claude分析研究笔记"""
    message = client.messages.create(
        model="claude-sonnet-4-6-20250514",
        max_tokens=1000,
        messages=[{
            "role": "user",
            "content": f"分析以下研究笔记，找出关键发现和潜在联系：\n\n{notes_content}"
        }]
    )
    return message.content[0].text
```
