```python
    """
    运行设计域
    """
    def __init__(self):
        self.dimensions = {
            'road_type': [],          # 道路类型
            'speed_range': (),        # 速度范围

        """
        定义道路类型
        """

            'highway',       # 高速公路
            'urban_arterial', # 城市主干道
            'urban_local',   # 城市次干道
            'residential',   # 居住区道路
            'parking',       # 停车场
            'construction'   # 施工区域
        ]
        self.dimensions['road_type'] = [t for t in road_types if t in valid_types]
    
    def define_speed_range(self, min_speed, max_speed):
        """
        定义速度范围
        """
        self.dimensions['speed_range'] = (min_speed, max_speed)
    
    def define_weather(self, conditions):
        """
        定义天气条件
        """
        valid_conditions = [
            'clear',        # 晴天
            'cloudy',       # 多云
            'rain_light',   # 小雨
            'rain_heavy',   # 大雨
            'snow_light',   # 小雪
            'snow_heavy',   # 大雪
            'fog',          # 雾天
            'night'         # 夜间
        ]
        self.dimensions['weather'] = [c for c in conditions if c in valid_conditions]
    
    def is_within_odd(self, current_conditions):
        """
        判断当前条件是否在ODD内
        """
        for dim, value in current_conditions.items():
            if dim in self.dimensions:
                allowed = self.dimensions[dim]
                if isinstance(allowed, list) and value not in allowed:
                    return False
                if isinstance(allowed, tuple):
                    if not (allowed[0] <= value <= allowed[1]):
                        return False
        return True
```

```python
class LiabilityFramework:
    """
    自动驾驶责任归属框架
    """
    def __init__(self):
        self.responsible_parties = {
            'driver': '人类驾驶员',
            'manufacturer': '汽车制造商',
            'software_developer': '软件开发商',
            'component_supplier': '零部件供应商',
            'fleet_operator': '车队运营商',
            'infrastructure_operator': '基础设施运营商'
        }
    
    def determine_liability(self, accident_info):
        """
        确定事故责任
        """
        automation_level = accident_info['automation_level']
        
        if automation_level in ['L0', 'L1', 'L2']:
            # 驾驶员主导
            liability = self.determine_driver_liability(accident_info)
        elif automation_level == 'L3':
            # 混合责任
            liability = self.determine_mixed_liability(accident_info)
        elif automation_level in ['L4', 'L5']:
            # 制造商/运营商主导
            liability = self.determine_manufacturer_liability(accident_info)
        
        return liability
    
    def determine_driver_liability(self, accident_info):
        """
        确定驾驶员责任
        """
        factors = {
            'attention_level': accident_info.get('driver_attention', 'attentive'),
            'intervention_opportunity': accident_info.get('intervention_time', 0),
            'system_warning': accident_info.get('warning_received', False)
        }
        
        if factors['attention_level'] == 'distracted':
            liability = 'primary_driver'
        elif not factors['system_warning']:
            liability = 'shared_manufacturer'
        else:
            liability = 'primary_driver'
        
        return liability
    
    def determine_mixed_liability(self, accident_info):
        """
        确定L3混合责任
        """
        takeover_request = accident_info.get('takeover_request_time', 0)
        takeover_response = accident_info.get('driver_response_time', 0)
        
        if takeover_response > 10:  # 秒
            return 'primary_driver'
        elif takeover_request < 3:  # 秒
            return 'primary_manufacturer'
        else:
            return 'shared'
    
    def determine_manufacturer_liability(self, accident_info):
        """
        确定制造商责任
        """
        system_malfunction = accident_info.get('system_malfunction', False)
        within_odd = accident_info.get('within_odd', True)
        
        if system_malfunction:
            return 'primary_manufacturer'
        elif not within_odd:
            return 'primary_fleet_operator'
        else:
            return 'primary_manufacturer'
```

```python
                'level_discount': level_factor,
```

```python
class TrolleyProblemSolver:
    """
    电车难题求解器
    """
    def __init__(self):
        self.ethical_frameworks = {
            'utilitarian': self.utilitarian_framework,
            'deontological': self.deontological_framework,
            'virtue_ethics': self.virtue_ethics_framework
        }
    
    def utilitarian_framework(self, scenario):
        """
        功利主义框架：最大化总体福祉
        """
        outcome_a = self.calculate_utility(scenario['option_a'])
        outcome_b = self.calculate_utility(scenario['option_b'])
        
        if outcome_a > outcome_b:
            return 'option_a'
        else:
            return 'option_b'
    
    def calculate_utility(self, option):
        """
        计算效用值
        """
        # 生命价值（等权重）
        lives_saved = option.get('lives_saved', 0)
        lives_lost = option.get('lives_lost', 0)
        
        # 伤害严重程度
        injury_severity = option.get('injury_severity', 0)
        
        utility = lives_saved * 100 - lives_lost * 100 - injury_severity * 10
        
        return utility
    
    def deontological_framework(self, scenario):
        """
        义务论框架：遵循道德规则
        """
        rules = {
            'do_not_harm': True,      # 不主动伤害
            'do_not_sacrifice': True, # 不主动牺牲
            'protect_occupant': True  # 保护乘员
        }
        
        # 主动转向导致伤害 vs 被动不作为
        if scenario.get('active_intervention', False):
            return 'do_not_intervene'  # 不主动干预
        else:
            return 'maintain_course'
    
    def virtue_ethics_framework(self, scenario):
        """
        美德伦理框架：考虑情境和品格
```

```python
class FairnessEvaluator:
    """
    算法公平性评估器
    """
    def __init__(self):
        self.protected_attributes = [
            'age', 'gender', 'race', 'disability'
        ]
    
    def evaluate_detection_fairness(self, detection_results, demographics):
        """
        评估目标检测公平性
        """
        fairness_metrics = {}
        
        for attr in self.protected_attributes:
            groups = demographics[attr].unique()
            group_metrics = {}
            
            for group in groups:
                group_mask = demographics[attr] == group
                group_results = detection_results[group_mask]
                
                group_metrics[group] = {
                    'detection_rate': self.calc_detection_rate(group_results),
                    'false_positive_rate': self.calc_fpr(group_results),
                    'false_negative_rate': self.calc_fnr(group_results)
                }
            
            # 计算公平性指标
            fairness_metrics[attr] = {
                'demographic_parity': self.calc_demographic_parity(group_metrics),
                'equalized_odds': self.calc_equalized_odds(group_metrics),
                'disparate_impact': self.calc_disparate_impact(group_metrics)
            }
        
        return fairness_metrics
    
    def calc_demographic_parity(self, group_metrics):
        """
        计算人口统计均等性
        """
        rates = [m['detection_rate'] for m in group_metrics.values()]
        return max(rates) - min(rates)
    
    def calc_equalized_odds(self, group_metrics):
        """
        计算均等机会
        """
        tpr_diff = max([m['detection_rate'] for m in group_metrics.values()]) - \
                   min([m['detection_rate'] for m in group_metrics.values()])
        fpr_diff = max([m['false_positive_rate'] for m in group_metrics.values()]) - \
                   min([m['false_positive_rate'] for m in group_metrics.values()])
        
        return {'tpr_difference': tpr_diff, 'fpr_difference': fpr_diff}
    
    def calc_disparate_impact(self, group_metrics):
        """
        计算差异影响比
        """
        rates = [m['detection_rate'] for m in group_metrics.values()]
        return min(rates) / max(rates) if max(rates) > 0 else 0
```

```python
class DataEthicsReview:
    """
    数据伦理审查
    """
    def __init__(self):
        self.review_criteria = {
            'consent': self.check_consent,
            'purpose_limitation': self.check_purpose,
            'data_minimization': self.check_minimization,
            'transparency': self.check_transparency,
            'accountability': self.check_accountability
        }
    
    def review(self, data_collection_plan):
        """
        审查数据收集计划
        """
        results = {}
        
        for criterion, check_func in self.review_criteria.items():
            results[criterion] = check_func(data_collection_plan)
        
        overall_pass = all(results.values())
        
        return {
            'overall_pass': overall_pass,
            'criteria_results': results,
            'recommendations': self.generate_recommendations(results)
        }
    
    def check_consent(self, plan):
        """
        检查知情同意
        """
        return plan.get('consent_mechanism') is not None
    
    def check_purpose(self, plan):
        """
        检查目的限制
        """
        return len(plan.get('purposes', [])) > 0
    
    def check_minimization(self, plan):
        """
        检查数据最小化
        """
        return plan.get('data_fields', []) <= plan.get('required_fields', [])
    
    def check_transparency(self, plan):
        """
        检查透明度
        """
        return plan.get('privacy_policy') is not None
    
    def check_accountability(self, plan):
        """
        检查问责机制
        """
        return plan.get('dpo_contact') is not None
    
    def generate_recommendations(self, results):
        """
        生成建议
        """
        recommendations = []
        
        if not results['consent']:
            recommendations.append('建议添加明确的知情同意机制')
        if not results['purpose_limitation']:
            recommendations.append('建议明确数据收集目的')
        if not results['data_minimization']:
            recommendations.append('建议减少不必要的数据收集字段')
```

```python
                'intervention_preference': self.calc_preference(results, 'intervene')
            }
        
        return analysis
    
    def calc_preference(self, results, preference_type):
        """
        计算偏好比例
        """
        matching = sum(1 for r in results if r['choice'] == preference_type)
        return matching / len(results) if len(results) > 0 else 0
```

```python
class EthicsReviewBoard:
    """
    伦理审查委员会
    """
    def __init__(self):
        self.committee_members = []
        self.review_history = []
    
    def submit_for_review(self, ethical_dilemma):
        """
        提交伦理审查
        """
        review_case = {
            'case_id': self.generate_case_id(),
            'dilemma': ethical_dilemma,
            'status': 'pending',
            'submissions': [],
            'decision': None
        }
        
        self.review_history.append(review_case)
        return review_case['case_id']
    
    def collect_opinions(self, case_id):
        """
        收集委员意见
        """
        opinions = []
        
        for member in self.committee_members:
            opinion = member.review(case_id)
            opinions.append({
                'member': member.name,
                'opinion': opinion,
                'reasoning': opinion.get('reasoning', '')
            })
        
        return opinions
    
    def make_decision(self, case_id):
        """
        做出决策
        """
        opinions = self.collect_opinions(case_id)
        
        # 多数决
        vote_counts = {}
        for opinion in opinions:
            decision = opinion['opinion']['decision']
            vote_counts[decision] = vote_counts.get(decision, 0) + 1
        
        final_decision = max(vote_counts, key=vote_counts.get)
        
        # 记录决策
        for case in self.review_history:
            if case['case_id'] == case_id:
                case['decision'] = final_decision
                case['status'] = 'resolved'
                break
        
        return {
            'decision': final_decision,
            'vote_distribution': vote_counts,
            'dissenting_opinions': [
                o for o in opinions if o['opinion']['decision'] != final_decision
            ]
        }
```

```python
class ChinaDataSecurityCompliance:
    """
    中国数据安全法合规
    """
    def __init__(self):
        self.data_classification = {
            'general_data': '一般数据',
            'important_data': '重要数据',
            'core_data': '核心数据'
        }
    
    def classify_data(self, data_info):
        """
        数据分类
        """
        classification_factors = {
            'volume': data_info.get('volume', 0),
            'sensitivity': data_info.get('sensitivity', 'low'),
            'scope': data_info.get('scope', 'internal'),
            'impact': data_info.get('impact', 'low')
        }
        
        if classification_factors['sensitivity'] == 'high' and \
           classification_factors['impact'] == 'high':
            return 'core_data'
        elif classification_factors['sensitivity'] == 'medium' or \
             classification_factors['volume'] > 1000000:
            return 'important_data'
        else:
            return 'general_data'
    
    def assess_cross_border_transfer(self, transfer_info):
        """
        评估跨境数据传输
        """
        requirements = {
            'security_assessment': False,
            'standard_contract': False,
            'certification': False
        }
        
        data_category = transfer_info['data_category']
        volume = transfer_info['volume']
        
        # 重要数据和核心数据需要安全评估
        if data_category in ['important_data', 'core_data']:
            requirements['security_assessment'] = True
        
        # 大规模个人信息需要标准合同
        if volume > 100000:
            requirements['standard_contract'] = True
        
        return requirements
```
