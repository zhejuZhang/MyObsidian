# 运动控制与轨迹跟踪

> 运动控制与轨迹跟踪是自动驾驶实现精确行驶的关键技术，将规划的轨迹转化为实际的车辆控制指令。

---

## 一、基本概念

### 1.1 运动控制定义
运动控制是将规划的轨迹转化为车辆控制指令的过程，包括：
- **纵向控制**：控制车辆的加速和制动
- **横向控制**：控制车辆的转向
- **协调控制**：协调纵向和横向控制

### 1.2 轨迹跟踪定义
轨迹跟踪是使车辆按照规划轨迹行驶的过程，包括：
- **位置跟踪**：跟踪目标位置
- **速度跟踪**：跟踪目标速度
- **姿态跟踪**：跟踪目标姿态

### 1.3 在自动驾驶中的作用

#### 精确控制
- **轨迹跟踪**：精确跟踪规划轨迹
- **速度控制**：精确控制行驶速度
- **姿态控制**：精确控制车辆姿态

#### 安全保障
- **稳定性控制**：保证车辆稳定性
- **舒适性控制**：保证乘坐舒适性
- **安全性控制**：保证行驶安全性

## 二、车辆动力学模型

### 2.1 运动学模型

#### 自行车模型
```python
class BicycleModel:
    """
    自行车运动学模型
    """
    def __init__(self, wheelbase=2.7):
        self.wheelbase = wheelbase  # 轴距
    
    def update(self, state, control, dt):
        """
        更新状态
        """
        x, y, theta, v = state
        a, delta = control
        
        # 运动学方程
        x_new = x + v * math.cos(theta) * dt
        y_new = y + v * math.sin(theta) * dt
        theta_new = theta + v / self.wheelbase * math.tan(delta) * dt
        v_new = v + a * dt
        
        return (x_new, y_new, theta_new, v_new)
```

### 2.2 动力学模型

#### 车辆动力学模型
```python
class VehicleDynamicsModel:
    """
    车辆动力学模型
    """
    def __init__(self):
        self.m = 1500  # 质量
        self.Iz = 2500  # 转动惯量
        self.lf = 1.2  # 前轴距
        self.lr = 1.4  # 后轴距
        self.Cf = 80000  # 前轮侧偏刚度
        self.Cr = 80000  # 后轮侧偏刚度
    
    def update(self, state, control, dt):
        """
        更新状态
        """
        x, y, theta, vx, vy, omega = state
        delta, Fx = control
        
        # 动力学方程
        alpha_f = delta - math.atan2(vy + self.lf * omega, vx)
        alpha_r = -math.atan2(vy - self.lr * omega, vx)
        
        Fyf = self.Cf * alpha_f
        Fyr = self.Cr * alpha_r
        
        ax = Fx / self.m - omega * vy
        ay = (Fyf * math.cos(delta) + Fyr) / self.m - omega * vx
        omega_dot = (self.lf * Fyf * math.cos(delta) - self.lr * Fyr) / self.Iz
        
        # 更新状态
        x_new = x + (vx * math.cos(theta) - vy * math.sin(theta)) * dt
        y_new = y + (vx * math.sin(theta) + vy * math.cos(theta)) * dt
        theta_new = theta + omega * dt
        vx_new = vx + ax * dt
        vy_new = vy + ay * dt
        omega_new = omega + omega_dot * dt
        
        return (x_new, y_new, theta_new, vx_new, vy_new, omega_new)
```

## 三、横向控制

### 3.1 Pure Pursuit控制器

#### Pure Pursuit算法
```python
class PurePursuitController:
    """
    Pure Pursuit横向控制器
    """
    def __init__(self, wheelbase=2.7):
        self.wheelbase = wheelbase
        self.lookahead_distance = 5.0  # 前视距离
    
    def control(self, current_pose, path, current_speed):
        """
        计算转向角
        """
        # 找到前视点
        lookahead_point = self.find_lookahead_point(current_pose, path)
        
        # 计算转向角
        alpha = math.atan2(
            lookahead_point[1] - current_pose[1],
            lookahead_point[0] - current_pose[0]
        ) - current_pose[2]
        
        # Pure Pursuit公式
        delta = math.atan2(
            2 * self.wheelbase * math.sin(alpha),
            self.lookahead_distance
        )
        
        return delta
    
    def find_lookahead_point(self, current_pose, path):
        """
        找到前视点
        """
        for point in path:
            distance = math.sqrt(
                (point[0] - current_pose[0]) ** 2 +
                (point[1] - current_pose[1]) ** 2
            )
            if distance >= self.lookahead_distance:
                return point
        
        return path[-1]
```

### 3.2 Stanley控制器

#### Stanley算法
```python
class StanleyController:
    """
    Stanley横向控制器
    """
    def __init__(self, wheelbase=2.7, k=0.5):
        self.wheelbase = wheelbase
        self.k = k  # 增益参数
    
    def control(self, current_pose, path, current_speed):
        """
        计算转向角
        """
        # 找到最近点
        nearest_point, nearest_index = self.find_nearest_point(
            current_pose, path
        )
        
        # 计算横向误差
        cross_track_error = self.compute_cross_track_error(
            current_pose, path, nearest_index
        )
        
        # 计算航向误差
        heading_error = self.compute_heading_error(
            current_pose, path, nearest_index
        )
        
        # Stanley公式
        delta = heading_error + math.atan2(
            self.k * cross_track_error,
            current_speed + 0.1
        )
        
        return delta
    
    def compute_cross_track_error(self, current_pose, path, nearest_index):
        """
        计算横向误差
        """
        # 计算到最近路径点的距离
        nearest_point = path[nearest_index]
        
        # 计算横向误差
        dx = current_pose[0] - nearest_point[0]
        dy = current_pose[1] - nearest_point[1]
        
        # 路径方向
        path_direction = math.atan2(
            path[nearest_index + 1][1] - path[nearest_index][1],
            path[nearest_index + 1][0] - path[nearest_index][0]
        )
        
        # 投影到法向量
        cross_track_error = dx * math.sin(path_direction) - dy * math.cos(path_direction)
        
        return cross_track_error
```

### 3.3 LQR横向控制器

#### LQR算法
```python
class LQRController:
    """
    LQR横向控制器
    """
    def __init__(self):
        # 状态权重矩阵
        self.Q = np.diag([1.0, 1.0, 1.0, 1.0])
        
        # 控制权重矩阵
        self.R = np.diag([1.0])
    
    def control(self, current_state, reference_state, dt):
        """
        计算控制量
        """
        # 线性化模型
        A, B = self.linearize_model(current_state, dt)
        
        # 求解Riccati方程
        P = self.solve_riccati(A, B, self.Q, self.R)
        
        # 计算LQR增益
        K = np.linalg.inv(self.R) @ B.T @ P
        
        # 计算控制量
        error = current_state - reference_state
        u = -K @ error
        
        return u[0]
    
    def linearize_model(self, state, dt):
        """
        线性化模型
        """
        # 线性化自行车模型
        A = np.eye(4)
        A[0, 2] = dt
        A[1, 3] = dt
        
        B = np.zeros((4, 1))
        B[2, 0] = dt
        
        return A, B
    
    def solve_riccati(self, A, B, Q, R):
        """
        求解Riccati方程
        """
        # 迭代求解
        P = Q
        for _ in range(100):
            P_new = Q + A.T @ P @ A - A.T @ P @ B @ np.linalg.inv(R + B.T @ P @ B) @ B.T @ P @ A
            
            if np.allclose(P, P_new, atol=1e-6):
                break
            
            P = P_new
        
        return P
```

## 四、纵向控制

### 4.1 PID纵向控制器

#### PID算法
```python
class PIDLongitudinalController:
    """
    PID纵向控制器
    """
    def __init__(self):
        self.kp = 1.0  # 比例增益
        self.ki = 0.1  # 积分增益
        self.kd = 0.01  # 微分增益
        self.integral = 0
        self.prev_error = 0
    
    def control(self, current_speed, target_speed, dt):
        """
        计算加速度
        """
        # 计算误差
        error = target_speed - current_speed
        
        # 积分项
        self.integral += error * dt
        
        # 微分项
        derivative = (error - self.prev_error) / dt
        
        # PID输出
        acceleration = (
            self.kp * error +
            self.ki * self.integral +
            self.kd * derivative
        )
        
        # 更新误差
        self.prev_error = error
        
        # 限制加速度范围
        acceleration = np.clip(acceleration, -4.0, 2.0)
        
        return acceleration
```

### 4.2 前馈+反馈控制

#### 前馈反馈控制器

```python
class FeedforwardFeedbackController:
    """
    前馈反馈纵向控制器
    """
    def __init__(self):
        self.feedback_controller = PIDLongitudinalController()
    
    def control(self, current_speed, target_speed, target_acceleration, dt):
        """
        计算加速度
        """
        # 前馈控制
        feedforward = target_acceleration
        
        # 反馈控制
        feedback = self.feedback_controller.control(
            current_speed, target_speed, dt
        )
        
        # 组合控制
        acceleration = feedforward + feedback
        
        return acceleration
```

## 五、模型预测控制（MPC）

### 5.1 线性MPC

#### 线性MPC控制器
```python
class LinearMPC:
    """
    线性MPC控制器
    """
    def __init__(self, horizon=20):
        self.horizon = horizon
        self.dt = 0.1
        
        # 系统矩阵
        self.A = None
        self.B = None
        
        # 权重矩阵
        self.Q = None
        self.R = None
        self.Qf = None
    
    def control(self, current_state, reference_trajectory):
        """
        计算控制序列
        """
        # 构建QP问题
        H, f, A_eq, b_eq, A_ineq, b_ineq = self.build_qp(
            current_state, reference_trajectory
        )
        
        # 求解QP
        solution = self.solve_qp(H, f, A_eq, b_eq, A_ineq, b_ineq)
        
        # 提取控制量
        control_sequence = solution[:self.horizon]
        
        return control_sequence[0]
    
    def build_qp(self, current_state, reference_trajectory):
        """
        构建QP问题
        """
        n_states = len(current_state)
        n_controls = 1  # 假设单输入
        
        # 决策变量：[u0, u1, ..., u_{N-1}, x0, x1, ..., x_N]
        n_vars = self.horizon * n_controls + (self.horizon + 1) * n_states
        
        # 目标函数
        H = np.zeros((n_vars, n_vars))
        f = np.zeros(n_vars)
        
        # 状态权重
        for i in range(self.horizon + 1):
            idx = self.horizon * n_controls + i * n_states
            H[idx:idx+n_states, idx:idx+n_states] = self.Q
        
        # 控制权重
        for i in range(self.horizon):
            idx = i * n_controls
            H[idx:idx+n_controls, idx:idx+n_controls] = self.R
        
        # 约束
        # ... 约束构建
        
        return H, f, A_eq, b_eq, A_ineq, b_ineq
```

## 六、协调控制

### 6.1 纵向横向协调

#### 协调控制器
```python
class CoordinatedController:
    """
    纵向横向协调控制器
    """
    def __init__(self):
        self.lateral_controller = LQRController()
        self.longitudinal_controller = PIDLongitudinalController()
    
    def control(self, current_state, reference_state, dt):
        """
        协调控制
        """
        # 横向控制
        steering_angle = self.lateral_controller.control(
            current_state, reference_state, dt
        )
        
        # 纵向控制
        acceleration = self.longitudinal_controller.control(
            current_state[3], reference_state[3], dt
        )
        
        # 协调调整
        steering_angle, acceleration = self.coordinate(
            steering_angle, acceleration, current_state
        )
        
        return steering_angle, acceleration
    
    def coordinate(self, steering_angle, acceleration, current_state):
        """
        协调调整
        """
        # 高速时减小转向角
        speed = current_state[3]
        if speed > 10:
            steering_angle *= 10.0 / speed
        
        # 大转向时减小加速度
        if abs(steering_angle) > 0.1:
            acceleration *= 0.5
        
        return steering_angle, acceleration
```

### 6.2 轨迹跟踪控制器

#### 轨迹跟踪器
```python
class TrajectoryTracker:
    """
    轨迹跟踪控制器
    """
    def __init__(self):
        self.controller = CoordinatedController()
        self.trajectory_index = 0
    
    def track(self, current_state, trajectory, dt):
        """
        跟踪轨迹
        """
        # 找到目标点
        target_state = self.find_target_state(current_state, trajectory)
        
        # 计算控制量
        steering_angle, acceleration = self.controller.control(
            current_state, target_state, dt
        )
        
        return steering_angle, acceleration
    
    def find_target_state(self, current_state, trajectory):
        """
        找到目标状态
        """
        # 找到最近点
        min_distance = float('inf')
        nearest_index = 0
        
        for i, point in enumerate(trajectory):
            distance = math.sqrt(
                (point[0] - current_state[0]) ** 2 +
                (point[1] - current_state[1]) ** 2
            )
            if distance < min_distance:
                min_distance = distance
                nearest_index = i
        
        # 前视目标
        lookahead_index = min(nearest_index + 5, len(trajectory) - 1)
        
        return trajectory[lookahead_index]
```
