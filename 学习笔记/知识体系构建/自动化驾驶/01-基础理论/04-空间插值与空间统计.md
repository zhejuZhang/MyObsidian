# 空间插值与空间统计

> 空间插值与空间统计是地理信息科学的核心方法，为自动驾驶中的环境建模、数据分析和决策支持提供理论基础。

---

## 一、基本概念

### 1.1 空间插值的定义
空间插值是根据已知点的值估计未知点值的方法，基于地理学第一定律：相近的事物更相似。

### 1.2 空间统计的定义
空间统计是分析空间数据分布模式和关联关系的统计方法，考虑空间位置信息。

### 1.3 在自动驾驶中的应用

#### 环境建模
- **地形建模**：插值生成连续地形表面
- **属性估计**：估计未采样区域的属性值
- **不确定性量化**：量化插值结果的不确定性

#### 数据分析
- **空间模式识别**：识别空间数据的分布模式
- **异常检测**：检测空间异常值
- **趋势分析**：分析空间数据的趋势

## 二、空间插值方法

### 2.1 反距离加权法（IDW）

#### 基本原理
```python
def idw_interpolation(points, values, target_point, power=2):
    """
    反距离加权插值
    """
    numerator = 0
    denominator = 0
    
    for i, point in enumerate(points):
        distance = euclidean_distance(point, target_point)
        if distance == 0:
            return values[i]
        
        weight = 1 / (distance ** power)
        numerator += weight * values[i]
        denominator += weight
    
    return numerator / denominator
```

#### 特点
- **优点**：简单直观，计算快速
- **缺点**：不考虑空间相关性，插值表面不平滑
- **适用**：数据分布均匀的情况

### 2.2 克里金法（Kriging）

#### 普通克里金
```python
class OrdinaryKriging:
    """
    普通克里金插值
    """
    def __init__(self, variogram_model='spherical'):
        self.variogram_model = variogram_model
        self.variogram_params = None
    
    def fit(self, points, values):
        """
        拟合变异函数
        """
        # 计算实验变异函数
        experimental_variogram = self.compute_experimental_variogram(points, values)
        # 拟合理论变异函数
        self.variogram_params = self.fit_variogram(experimental_variogram)
    
    def predict(self, target_point):
        """
        预测目标点的值
        """
        # 构建克里金方程组
        n = len(self.points)
        K = np.zeros((n + 1, n + 1))
        
        # 填充变异函数矩阵
        for i in range(n):
            for j in range(n):
                K[i, j] = self.variogram(self.points[i], self.points[j])
        
        # 添加约束条件
        K[n, :n] = 1
        K[:n, n] = 1
        K[n, n] = 0
        
        # 构建右侧向量
        k = np.zeros(n + 1)
        for i in range(n):
            k[i] = self.variogram(self.points[i], target_point)
        k[n] = 1
        
        # 求解权重
        weights = np.linalg.solve(K, k)
        
        # 计算预测值
        prediction = sum(weights[i] * self.values[i] for i in range(n))
        
        # 计算方差
        variance = sum(weights[i] * k[i] for i in range(n + 1))
        
        return prediction, variance
```

#### 变异函数模型
```python
def spherical_variogram(h, nugget, sill, range_param):
    """
    球状变异函数模型
    """
    if h == 0:
        return 0
    elif h <= range_param:
        return nugget + (sill - nugget) * (1.5 * h / range_param - 0.5 * (h / range_param) ** 3)
    else:
        return sill

def exponential_variogram(h, nugget, sill, range_param):
    """
    指数变异函数模型
    """
    return nugget + (sill - nugget) * (1 - np.exp(-3 * h / range_param))

def gaussian_variogram(h, nugget, sill, range_param):
    """
    高斯变异函数模型
    """
    return nugget + (sill - nugget) * (1 - np.exp(-3 * (h / range_param) ** 2))
```

- **普通克里金**：未知均值
- **泛克里金**：考虑趋势
- **协同克里金**：多变量协同

```python
def thin_plate_spline(points, values, target_point, smoothing=0):
    """
    薄板样条插值
    """
    n = len(points)
    
    # 构建径向基函数矩阵
    K = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            r = euclidean_distance(points[i], points[j])
            if r == 0:
                K[i, j] = 0
            else:
                K[i, j] = r ** 2 * np.log(r)
    
    # 添加平滑项
    K += smoothing * np.eye(n)
    
    # 求解权重
    weights = np.linalg.solve(K, values)
    
    # 计算预测值
    prediction = 0
    for i in range(n):
        r = euclidean_distance(points[i], target_point)
        if r > 0:
            prediction += weights[i] * r ** 2 * np.log(r)
    
    return prediction
```

## 三、空间统计方法

### 3.1 空间自相关分析

#### 全局Moran's I
```python
def global_morans_i(values, weights):
    """
    全局Moran's I检验
    """
    n = len(values)
    mean = np.mean(values)
    
    # 计算分子
    numerator = 0
    for i in range(n):
        for j in range(n):
            numerator += weights[i][j] * (values[i] - mean) * (values[j] - mean)
    
    # 计算分母
    denominator = sum((values - mean) ** 2)
    
    # 计算Moran's I
    W = np.sum(weights)
    I = (n / W) * (numerator / denominator)
    
    # 计算期望和方差
    E_I = -1 / (n - 1)
    
    return I, E_I
```

#### 局部Moran's I（LISA）
```python
def local_morans_i(values, weights):
    """
    局部Moran's I（LISA）
    """
    n = len(values)
    mean = np.mean(values)
    std = np.std(values)
    
    I_local = []
    for i in range(n):
        # 标准化
        zi = (values[i] - mean) / std
        
        # 计算邻域加权和
        lag = sum(weights[i][j] * (values[j] - mean) for j in range(n)) / std
        
        # 计算局部Moran's I
        I_i = zi * lag
        I_local.append(I_i)
    
    return I_local
```

- **高低异常**：高值被低值包围

#### 地理加权回归（GWR）
```python
class GeographicallyWeightedRegression:
    """
    地理加权回归
    """
    def __init__(self, bandwidth='adaptive'):
        self.bandwidth = bandwidth
        self.kernel = 'gaussian'
    
    def fit(self, X, y, coordinates):
        """
        拟合地理加权回归模型
        """
        n = len(y)
        self.coefficients = np.zeros((n, X.shape[1]))
        
        for i in range(n):
            # 计算权重
            weights = self.compute_weights(coordinates, coordinates[i])
            
            # 加权最小二乘
            W = np.diag(weights)
            XtWX = X.T @ W @ X
            XtWy = X.T @ W @ y
            self.coefficients[i] = np.linalg.solve(XtWX, XtWy)
    
    def compute_weights(self, coordinates, target_coord):
        """
        计算空间权重
        """
        distances = np.sqrt(np.sum((coordinates - target_coord) ** 2, axis=1))
        
        if self.kernel == 'gaussian':
            weights = np.exp(-0.5 * (distances / self.bandwidth) ** 2)
        elif self.kernel == 'bisquare':
            weights = np.where(distances <= self.bandwidth, 
                              (1 - (distances / self.bandwidth) ** 2) ** 2, 0)
        
        return weights
```

### 3.3 空间点模式分析

#### 最近邻分析
```python
def nearest_neighbor_analysis(points):
    """
    最近邻分析
    """
    n = len(points)
    
    # 计算每个点的最近邻距离
    nn_distances = []
    for i in range(n):
        min_dist = float('inf')
        for j in range(n):
            if i != j:
                dist = euclidean_distance(points[i], points[j])
                min_dist = min(min_dist, dist)
        nn_distances.append(min_dist)
    
    # 计算平均最近邻距离
    mean_nn = np.mean(nn_distances)
    
    # 计算期望最近邻距离
    area = compute_area(points)
    density = n / area
    expected_nn = 0.5 / np.sqrt(density)
    
    # 计算最近邻指数
    R = mean_nn / expected_nn
    
    return R, mean_nn, expected_nn
```

#### K函数分析
```python
def k_function(points, max_distance, n_bins=20):
    """
    K函数分析
    """
    n = len(points)
    area = compute_area(points)
    density = n / area
    
    # 计算距离范围
    distances = np.linspace(0, max_distance, n_bins + 1)
    K_values = []
    
    for d in distances:
        # 计算每个点在距离d内的邻居数
        count = 0
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist = euclidean_distance(points[i], points[j])
                    if dist <= d:
                        count += 1
        
        # 计算K函数值
        K = count / (n * density)
        K_values.append(K)
    
    return distances, K_values
```

```python
class DEMGenerator:
    """
    数字高程模型生成
    """
    def __init__(self, method='kriging'):
        self.method = method
    
    def generate(self, points, values, grid_resolution):
        """
        生成DEM
        """
        # 创建网格
        x_min, x_max = min(p[0] for p in points), max(p[0] for p in points)
        y_min, y_max = min(p[1] for p in points), max(p[1] for p in points)
        
        x_grid = np.arange(x_min, x_max, grid_resolution)
        y_grid = np.arange(y_min, y_max, grid_resolution)
        
        # 插值
        dem = np.zeros((len(y_grid), len(x_grid)))
        for i, y in enumerate(y_grid):
            for j, x in enumerate(x_grid):
                if self.method == 'idw':
                    dem[i, j] = idw_interpolation(points, values, (x, y))
                elif self.method == 'kriging':
                    dem[i, j], _ = self.kriging.predict((x, y))
        
        return dem
```

```python
def monte_carlo_uncertainty(kriging_model, target_point, n_simulations=1000):
    """
    蒙特卡洛不确定性量化
    """
    # 获取预测值和方差
    prediction, variance = kriging_model.predict(target_point)
    
    # 生成随机样本
    samples = np.random.normal(prediction, np.sqrt(variance), n_simulations)
    
    # 计算置信区间
    confidence_interval = np.percentile(samples, [2.5, 97.5])
    
    return prediction, confidence_interval
```

```python
def kernel_density_estimation(points, bandwidth, grid_resolution):
    """
    核密度估计
    """
    # 创建网格
    x_min, x_max = min(p[0] for p in points), max(p[0] for p in points)
    y_min, y_max = min(p[1] for p in points), max(p[1] for p in points)

    y_grid = np.arange(y_min, y_max, grid_resolution)

    density = np.zeros((len(y_grid), len(x_grid)))
    for i, y in enumerate(y_grid):
        for j, x in enumerate(x_grid):
            for point in points:
                dist = euclidean_distance((x, y), point)
                if dist <= bandwidth:
                    density[i, j] += gaussian_kernel(dist / bandwidth)
    
    return density
```

```python
class BayesianKriging:
    """
    贝叶斯克里金
    """
    def __init__(self, prior_distribution):
        self.prior = prior_distribution
        self.posterior = None
    
    def fit(self, points, values):
        """
        贝叶斯克里金拟合
        """
        # 计算似然
        likelihood = self.compute_likelihood(points, values)
        
        # 计算后验
        self.posterior = self.prior * likelihood
        self.posterior /= self.posterior.sum()
    
    def predict(self, target_point):
        """
        贝叶斯预测
        """
        # 从后验分布采样
        samples = self.sample_posterior(target_point)
        
        # 计算预测均值和方差
        prediction = np.mean(samples)
        variance = np.var(samples)
        
        return prediction, variance
```
