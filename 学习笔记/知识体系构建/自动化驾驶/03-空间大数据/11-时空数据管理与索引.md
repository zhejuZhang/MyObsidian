# 时空数据管理与索引

> 时空数据管理与索引是自动驾驶系统高效处理海量时空数据的关键技术。

---

## 一、基本概念

### 1.1 时空数据定义
时空数据是同时包含空间位置和时间信息的数据，具有：
- **空间维度**：地理位置坐标
- **时间维度**：时间戳或时间段
- **属性维度**：其他相关属性

### 1.2 时空数据特点

#### 数据特点
- **数据量大**：传感器产生海量数据
- **更新频繁**：数据持续更新
- **查询复杂**：时空查询复杂

#### 应用特点
- **实时性要求**：需要实时处理
- **历史查询**：需要查询历史数据
- **预测分析**：需要预测未来状态

### 1.3 在自动驾驶中的应用

#### 实时处理
- **传感器数据流**：处理实时传感器数据
- **目标跟踪**：跟踪移动目标
- **状态估计**：估计车辆状态

#### 历史分析
- **轨迹分析**：分析历史轨迹
- **模式识别**：识别运动模式
- **异常检测**：检测异常行为

## 二、时空数据模型

### 2.1 移动对象数据模型

#### 移动点模型
```python
class MovingPoint:
    """
    移动点数据模型
    """
    def __init__(self, object_id):
        self.object_id = object_id
        self.trajectory = []  # 轨迹点列表
    
    def add_position(self, timestamp, x, y, z=None):
        """
        添加位置点
        """
        self.trajectory.append({
            'timestamp': timestamp,
            'x': x,
            'y': y,
            'z': z
        })
    
    def get_position(self, timestamp):
        """
        获取指定时间的位置
        """
        # 插值计算位置
        return self.interpolate(timestamp)
    
    def interpolate(self, timestamp):
        """
        插值计算
        """
        # 找到前后两个点
        before, after = self.find_surrounding_points(timestamp)
        
        # 线性插值
        ratio = (timestamp - before['timestamp']) / \
                (after['timestamp'] - before['timestamp'])
        
        x = before['x'] + ratio * (after['x'] - before['x'])
        y = before['y'] + ratio * (after['y'] - before['y'])
        
        return (x, y)
```

### 2.2 时空立方体模型

#### 时空立方体
```python
class SpatioTemporalCube:
    """
    时空立方体
    """
    def __init__(self, spatial_resolution, temporal_resolution):
        self.spatial_res = spatial_resolution
        self.temporal_res = temporal_resolution
        self.cube = {}
    
    def insert(self, x, y, t, value):
        """
        插入数据
        """
        # 量化坐标
        grid_x = int(x / self.spatial_res)
        grid_y = int(y / self.spatial_res)
        grid_t = int(t / self.temporal_res)
        
        key = (grid_x, grid_y, grid_t)
        if key not in self.cube:
            self.cube[key] = []
        self.cube[key].append(value)
    
    def query(self, x_range, y_range, t_range):
        """
        时空范围查询
        """
        results = []
        
        # 量化查询范围
        x_min = int(x_range[0] / self.spatial_res)
        x_max = int(x_range[1] / self.spatial_res)
        y_min = int(y_range[0] / self.spatial_res)
        y_max = int(y_range[1] / self.spatial_res)
        t_min = int(t_range[0] / self.temporal_res)
        t_max = int(t_range[1] / self.temporal_res)
        
        # 遍历查询
        for x in range(x_min, x_max + 1):
            for y in range(y_min, y_max + 1):
                for t in range(t_min, t_max + 1):
                    key = (x, y, t)
                    if key in self.cube:
                        results.extend(self.cube[key])
        
        return results
```

### 3.1 R树及其变种

#### 3D R树
```python
class RTree3D:
    """
    三维R树索引
    """
    def __init__(self, max_entries=25):
        self.max_entries = max_entries
        self.root = None
    
    def insert(self, x, y, t, data):
        """
        插入时空数据
        """
        # 计算MBR（最小边界矩形）
        mbr = self.compute_mbr(x, y, t)
        
        # 插入到R树
        self._insert_recursive(self.root, mbr, data)
    
    def query(self, x_range, y_range, t_range):
        """
        时空范围查询
        """
        query_mbr = self.compute_query_mbr(x_range, y_range, t_range)
        results = []
        self._query_recursive(self.root, query_mbr, results)
        return results
```

#### TPR树（时间参数化R树）
```python
class TPRTree:
    """
    时间参数化R树
    """
    def __init__(self):
        self.root = None
    
    def insert(self, object_id, position, velocity):
        """
        插入移动对象
        """
        # 计算时间参数化MBR
        tp_mbr = self.compute_tp_mbr(position, velocity)
        
        # 插入到TPR树
        self._insert(self.root, tp_mbr, object_id)
    
    def query(self, query_region, query_time):
        """
        查询指定时间的区域内的对象
        """
        # 计算查询时间的MBR
        query_mbr = self.compute_query_mbr(query_region, query_time)
        
        # 查询
        results = []
        self._query(self.root, query_mbr, query_time, results)
        return results
```

### 3.2 网格索引

#### 时空网格索引
```python
class SpatioTemporalGrid:
    """
    时空网格索引
    """
    def __init__(self, cell_size, time_step):
        self.cell_size = cell_size
        self.time_step = time_step
        self.grid = {}
    
    def get_cell_key(self, x, y, t):
        """
        获取网格单元键值
        """
        cell_x = int(x / self.cell_size)
        cell_y = int(y / self.cell_size)
        cell_t = int(t / self.time_step)
        return (cell_x, cell_y, cell_t)
    
    def insert(self, x, y, t, data):
        """
        插入数据
        """
        key = self.get_cell_key(x, y, t)
        if key not in self.grid:
            self.grid[key] = []
        self.grid[key].append(data)
    
    def query_range(self, x_range, y_range, t_range):
        """
        范围查询
        """
        results = []
        
        # 计算查询的网格范围
        x_min_cell = int(x_range[0] / self.cell_size)
        x_max_cell = int(x_range[1] / self.cell_size)
        y_min_cell = int(y_range[0] / self.cell_size)
        y_max_cell = int(y_range[1] / self.cell_size)
        t_min_cell = int(t_range[0] / self.time_step)
        t_max_cell = int(t_range[1] / self.time_step)
        
        # 遍历查询
        for x in range(x_min_cell, x_max_cell + 1):
            for y in range(y_min_cell, y_max_cell + 1):
                for t in range(t_min_cell, t_max_cell + 1):
                    key = (x, y, t)
                    if key in self.grid:
                        results.extend(self.grid[key])
        
        return results
```

### 3.3 轨迹索引

#### 轨迹段索引
```python
class TrajectoryIndex:
    """
    轨迹索引
    """
    def __init__(self):
        self.trajectories = {}
        self.spatial_index = RTree()
        self.temporal_index = {}
    
    def add_trajectory(self, trajectory_id, points):
        """
        添加轨迹
        """
        self.trajectories[trajectory_id] = points
        
        # 建立空间索引
        for i, point in enumerate(points):
            self.spatial_index.insert(
                point['x'], point['y'], 
                {'trajectory_id': trajectory_id, 'point_index': i}
            )
        
        # 建立时间索引
        for point in points:
            t = point['timestamp']
            if t not in self.temporal_index:
                self.temporal_index[t] = []
            self.temporal_index[t].append(trajectory_id)
    
    def query_spatial(self, x_range, y_range):
        """
        空间查询
        """
        return self.spatial_index.query(x_range, y_range)
    
    def query_temporal(self, t_range):
        """
        时间查询
        """
        results = []
        for t in self.temporal_index:
            if t_range[0] <= t <= t_range[1]:
                results.extend(self.temporal_index[t])
        return list(set(results))
```

### 4.1 范围查询

#### 时空范围查询

class SpatioTemporalRangeQuery:
    """
    时空范围查询
    """
    def __init__(self, index):
        self.index = index
    
    def query(self, spatial_range, temporal_range):
        """
        执行时空范围查询
        """
        # 使用索引加速查询
        candidates = self.index.query_range(
            spatial_range, temporal_range
        )
        
        # 过滤和验证
        results = []
        for candidate in candidates:
            if self.validate(candidate, spatial_range, temporal_range):
                results.append(candidate)
        
        return results
```

#### 时空最近邻查询

class SpatioTemporalKNN:
    """
    时空K近邻查询
    """
    def __init__(self, index):
        self.index = index
    
    def query(self, query_point, query_time, k):
        """
        查询k个最近邻
        """
        # 计算距离
        distances = []
        for obj in self.index.get_all_objects():
            dist = self.compute_distance(
                query_point, query_time, obj
            )
            distances.append((dist, obj))
        
        # 排序并返回前k个
        distances.sort()
        return distances[:k]
```

### 5.2 分布式存储

#### 时空数据分区
```python
class SpatioTemporalPartitioner:
    """
    时空数据分区器
    """
    def __init__(self, spatial_grid, temporal_grid):
        self.spatial_grid = spatial_grid
        self.temporal_grid = temporal_grid
    
    def partition(self, data):
        """
        分区时空数据
        """
        partitions = {}
        
        for record in data:
            # 计算空间分区
            spatial_key = self.get_spatial_partition(
                record['x'], record['y']
            )
            
            # 计算时间分区
            temporal_key = self.get_temporal_partition(
                record['timestamp']
            )
            
            # 组合分区键
            partition_key = (spatial_key, temporal_key)
            
            if partition_key not in partitions:
                partitions[partition_key] = []
            partitions[partition_key].append(record)
        
        return partitions
```

## 六、实时流处理

### 6.1 流处理框架

#### 时间窗口处理
```python
class TemporalWindowProcessor:
    """
    时间窗口处理器
    """
    def __init__(self, window_size, slide_interval):
        self.window_size = window_size
        self.slide_interval = slide_interval
        self.buffer = []
    
    def process(self, data_stream):
        """
        处理数据流
        """
        results = []
        
        for data in data_stream:
            # 添加到缓冲区
            self.buffer.append(data)
            
            # 移除过期数据
            self.remove_expired(data['timestamp'])
            
            # 处理窗口内的数据
            if self.is_window_ready():
                window_result = self.process_window()
                results.append(window_result)
        
        return results
    
    def remove_expired(self, current_time):
        """
        移除过期数据
        """
        expired_time = current_time - self.window_size
        self.buffer = [
            data for data in self.buffer
            if data['timestamp'] > expired_time
        ]
```

### 6.2 增量计算

#### 增量聚合
```python
class IncrementalAggregator:
    """
    增量聚合器
    """
    def __init__(self):
        self.current_state = None
    
    def update(self, new_data):
        """
        增量更新
        """
        if self.current_state is None:
            self.current_state = self.initialize(new_data)
        else:
            self.current_state = self.merge(self.current_state, new_data)
        
        return self.current_state
    
    def initialize(self, data):
        """
        初始化状态
        """
        return {
            'count': 1,
            'sum': data['value'],
            'min': data['value'],
            'max': data['value']
        }
    
    def merge(self, state, data):
        """
        合并状态
        """
        return {
            'count': state['count'] + 1,
            'sum': state['sum'] + data['value'],
            'min': min(state['min'], data['value']),
            'max': max(state['max'], data['value'])
        }
```
